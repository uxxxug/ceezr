import { chromium, type Page } from "playwright-core";
import { readFileSync, mkdirSync } from "node:fs";
import { inspectChatMedia } from "../packages/domain/messaging/chat-media.ts";
const ar = JSON.parse(readFileSync("../packages/shared/i18n/miniapp/ar.json", "utf8")) as Record<string, string>;
const L = (k: string) => ar[`driver.job.chat.media.${k}`]!;
const exe = `${process.env.HOME}/.cache/ms-playwright/chromium-1217/chrome-linux64/chrome`;
const SHOTS = "/tmp/pw/shots"; mkdirSync(SHOTS, { recursive: true });
const files: Record<string, [string, string]> = {
  "/out/entry.js": ["out/entry.js", "text/javascript"], "/out/entry.css": ["out/entry.css", "text/css"],
  "/photo.jpg": ["../tests/fixtures/chat-media/photo-gps.jpg", "image/jpeg"],
  "/voice.webm": ["../tests/fixtures/chat-media/chromium-recorded.webm", "audio/webm"],
};
const server = Bun.serve({ port: 0, fetch: (req) => { const p = new URL(req.url).pathname; const f = files[p];
  if (f) return new Response(readFileSync(f[0]), { headers: { "content-type": f[1] } });
  return new Response(`<!doctype html><html><head><meta name=viewport content="width=device-width"><link rel=stylesheet href=/out/entry.css></head><body><div id=root></div><script type=module src=/out/entry.js></script></body></html>`, { headers: { "content-type": "text/html" } }); } });
const origin = `http://localhost:${server.port}`;
const report: Record<string, unknown> = {};
const errors: string[] = [];
async function open(args: string[], role: string): Promise<[Page, () => Promise<void>]> {
  const browser = await chromium.launch({ executablePath: exe, args });
  const ctx = await browser.newContext({ viewport: { width: 390, height: 844 }, deviceScaleFactor: 2, isMobile: true, hasTouch: true });
  const page = await ctx.newPage();
  page.on("pageerror", (e) => errors.push(`${role}: ${e.message}`));
  await page.goto(`${origin}/?role=${role}`);
  return [page, () => browser.close()];
}
{
  const [page, close] = await open(["--use-fake-device-for-media-stream", "--use-fake-ui-for-media-stream"], "driver");
  await page.getByText(L("driverSafetyNotice")).waitFor();
  await page.screenshot({ path: `${SHOTS}/01-driver-open-safety-notice.png`, fullPage: true });
  report.htmlDir = await page.evaluate(() => document.documentElement.dir);
  report.textareaDir = await page.locator("textarea.ocht__input").getAttribute("dir");
  await page.getByRole("button", { name: L("driverSafetyDismiss"), exact: true }).click();
  await page.locator("img").first().waitFor({ timeout: 5000 }).catch(() => undefined);
  await page.waitForTimeout(800);
  await page.screenshot({ path: `${SHOTS}/02-driver-thread-rtl.png`, fullPage: true });
  report.bubbleSides = await page.evaluate(() => [...document.querySelectorAll(".ocht__msg")].map((el) => { const r = el.getBoundingClientRect(); return { cls: el.className, left: Math.round(r.left), right: Math.round(r.right) }; }));
  // صوت
  await page.getByRole("button", { name: L("recordVoice"), exact: true }).click();
  await page.getByText(L("driverSafetyGate")).waitFor();
  await page.screenshot({ path: `${SHOTS}/03-driver-safety-gate.png`, fullPage: true });
  await page.getByRole("button", { name: L("driverSafetyConfirm"), exact: true }).click();
  await page.getByRole("button", { name: L("stopRecording"), exact: true }).waitFor({ timeout: 5000 });
  await page.waitForTimeout(2500);
  await page.screenshot({ path: `${SHOTS}/04-driver-recording.png`, fullPage: true });
  await page.getByRole("button", { name: L("stopRecording"), exact: true }).click();
  await page.getByRole("button", { name: L("sendVoice"), exact: true }).click();
  await page.waitForFunction(() => (window as unknown as { uploads: unknown[] }).uploads.length >= 1);
  // صورة
  await page.getByRole("button", { name: L("attach"), exact: true }).click();
  const chooser = page.waitForEvent("filechooser");
  await page.getByRole("button", { name: L("attachImage"), exact: true }).click();
  await (await chooser).setFiles("../tests/fixtures/chat-media/photo-gps.jpg");
  await page.getByRole("button", { name: L("sendImage"), exact: true }).click();
  await page.waitForFunction(() => (window as unknown as { uploads: unknown[] }).uploads.length >= 2);
  await page.waitForTimeout(800);
  await page.screenshot({ path: `${SHOTS}/05-driver-after-send.png`, fullPage: true });
  const uploads = (await page.evaluate(() => (window as unknown as { uploads: unknown[] }).uploads)) as { kind: "image" | "voice"; type: string; durationMs: number | null; b64: string }[];
  report.uploads = uploads.map((u) => { const bytes = Uint8Array.from(Buffer.from(u.b64, "base64")); const v = inspectChatMedia({ kind: u.kind, bytes, declaredDurationMs: u.durationMs ?? undefined });
    return { kind: u.kind, blobType: u.type, clientDurationMs: u.durationMs, bytes: bytes.length, verdict: v.ok ? { ok: true, mime: v.mime, ...(v.kind === "voice" ? { durationMs: v.durationMs } : { width: v.width, height: v.height, strippedBytes: v.strippedBytes }) } : v,
      exifInClientOutput: Buffer.from(bytes).includes(Buffer.from("Exif")) }; });
  await close();
}
{
  const [page, close] = await open(["--use-fake-device-for-media-stream", "--use-fake-ui-for-media-stream"], "rider");
  await page.waitForTimeout(1200);
  report.riderHasSafetyNotice = await page.locator(".ocht__safety").count();
  await page.screenshot({ path: `${SHOTS}/06-rider-thread-rtl.png`, fullPage: true });
  await close();
}
{
  // رفضُ الميكروفون: بلا fake-ui ⇒ Chromium بلا واجهةٍ يرفضُ الطلب.
  const [page, close] = await open(["--use-fake-device-for-media-stream", "--deny-permission-prompts"], "rider");
  await page.waitForTimeout(800);
  await page.getByRole("button", { name: ar["rider.active.chat.media.recordVoice"] }).click();
  await page.getByText(ar["rider.active.chat.media.micDenied"]!).waitFor({ timeout: 5000 });
  report.micDeniedShown = true;
  await page.screenshot({ path: `${SHOTS}/07-rider-mic-denied.png`, fullPage: true });
  await close();
}
server.stop();
report.pageErrors = errors;
console.log(JSON.stringify(report, null, 1));

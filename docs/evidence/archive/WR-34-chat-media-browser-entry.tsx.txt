  id, kind, mime_type: kind === "image" ? "image/jpeg" : "audio/webm", byte_size: 30000,
  width: kind === "image" ? 1200 : null, height: kind === "image" ? 900 : null,
  duration_ms: kind === "voice" ? 2400 : null, available: true, ...over,

const mine = role === "driver" ? "driver" : "rider";
const theirs = role === "driver" ? "rider" : "driver";
let seq = 4;
const messages: unknown[] = [
  { id: "1", seq: 1, sender_role: theirs, mine: false, client_message_id: null, body: "السلام عليكم، أنا عند البوابة الشمالية", created_at: "2026-11-03T10:00:00Z", kind: "text", location: null, translation: null },
  { id: "2", seq: 2, sender_role: mine, mine: true, client_message_id: "c2", body: "وعليكم السلام، دقيقتين وأوصل", created_at: "2026-11-03T10:01:00Z", kind: "text", location: null, translation: null },
  { id: "3", seq: 3, sender_role: theirs, mine: false, client_message_id: null, body: "📷", created_at: "2026-11-03T10:02:00Z", kind: "image", location: null, translation: null, attachment: att("a3", "image") },
  { id: "4", seq: 4, sender_role: mine, mine: true, client_message_id: "c4", body: "🎤", created_at: "2026-11-03T10:03:00Z", kind: "voice", location: null, translation: null, attachment: att("a4", "voice") },

  counterpart: { first_name: role === "driver" ? "سارة" : "سالم" }, messages, last_seq: seq,

w.uploads = [];

  fetchTail: async () => thread() as never,

  fetchBefore: async () => ({ ...thread(), messages: [], has_older: false }) as never,

    const buf = new Uint8Array(await media.blob.arrayBuffer());
    let bin = ""; for (const b of buf) bin += String.fromCharCode(b);
    w.uploads.push({ kind: media.kind, type: media.blob.type, durationMs: media.durationMs, b64: btoa(bin) });
    seq += 1;

    messages.push(m);
    return { status: "added", message: m } as never;
  },
  attachmentUrl: async (_o, id) => ({ url: id === "a3" ? "/photo.jpg" : "/voice.webm", expiresAt: new Date(Date.now() + 300000).toISOString(), mimeType: id === "a3" ? "image/jpeg" : "audio/webm" }),

  plugins: [{ name: "alias", setup(b) { b.onResolve({ filter: /^(react|react-dom|preact)(\/.*)?$/ }, (a) => ({ path: Bun.resolveSync(map[a.path] ?? a.path, `${process.cwd()}/../apps/miniapp`) })); } }] });
